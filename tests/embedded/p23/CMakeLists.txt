cmake_minimum_required(VERSION 3.16)

set(DMP_P23_PROFILE "" CACHE STRING "Measured P23 responder profile")
option(DMP_ENABLE_ENDPOINT_STREAM_R "Support Stream R binding at the endpoint" ON)
set_property(CACHE DMP_P23_PROFILE PROPERTY STRINGS direct-nnpsk0 radio-xx)
if(NOT DMP_P23_PROFILE MATCHES "^(direct-nnpsk0|radio-xx)$")
    message(FATAL_ERROR "Select DMP_P23_PROFILE=direct-nnpsk0 or radio-xx")
endif()
if(NOT IDF_TARGET MATCHES "^(esp32s3|esp32c3)$")
    message(FATAL_ERROR "P23 supports the ESP32-S3 and ESP32-C3 emulators")
endif()
if(DMP_HS_SHARED_ADMISSION_ONLY)
    message(FATAL_ERROR "P23 measures responder-owned handshake admission")
endif()

set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
set(COMPONENTS main)
get_filename_component(DMP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(EXTRA_COMPONENT_DIRS
    "${CMAKE_CURRENT_LIST_DIR}/../components/dmp"
    "${DMP_ROOT}/third_party/noise-c")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "${CMAKE_BINARY_DIR}/dependencies.lock")
project(dmp_p23_probe)
if(NOT IDF_VERSION_MAJOR EQUAL 6 OR NOT IDF_VERSION_MINOR EQUAL 1)
    message(FATAL_ERROR "P23 requires ESP-IDF 6.1")
endif()
if(DMP_P23_PROFILE STREQUAL "direct-nnpsk0" AND NOT DMP_ENABLE_ENDPOINT_STREAM_R)
    message(FATAL_ERROR "DIRECT-1 requires endpoint Stream R")
endif()
